# Use various selection commands (10)

	select acidic, resn ASP+GLU
	select basic, resn LYS+ARG
	select aromatic, resn PHE+TYR
	
	select nterm, resi 1-30
	select cterm, resi 200-300
	
	select nterm_acidic, nterm and acidic
	select ionized, acidic or basic

# Find active center by distance (10)

	select active, all within 6 of resn ZN
	select active_residues, byres active

	select active_ligand, all within 5 of resn AZM
	select active_ligand_residues, byres active_ligand

# Measure distance (5)

	distance d_zn_ligand, name ZN, resn AZM and name N1

	select zn2, 1ca2 and resn ZN

	align 1ca2, 1cam
	
	select zn3, 1cam and resn ZN
